cmake_minimum_required(VERSION 3.16)
project(VisionFlow VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
option(VISIONFLOW_ENABLE_ASAN "Address sanitizer" OFF)
option(VISIONFLOW_ENABLE_UBSAN "Undefined behavior sanitizer" OFF)
option(VISIONFLOW_ENABLE_TSAN "Thread sanitizer" OFF)
option(VISIONFLOW_CORE_ONLY "Build dependency-free protocol/concurrency tests only" OFF)
if(VISIONFLOW_ENABLE_ASAN AND VISIONFLOW_ENABLE_TSAN)
  message(FATAL_ERROR "ASan and TSan require separate build directories")
endif()
find_package(Threads REQUIRED)
add_library(visionflow_options INTERFACE)
target_compile_options(visionflow_options INTERFACE
  $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wshadow>)
foreach(kind ASAN UBSAN TSAN)
  if(VISIONFLOW_ENABLE_${kind})
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      message(FATAL_ERROR "Sanitizers require GCC or Clang")
    endif()
    if(kind STREQUAL "ASAN")
      set(sanitizer address)
    elseif(kind STREQUAL "UBSAN")
      set(sanitizer undefined)
    else()
      set(sanitizer thread)
    endif()
    target_compile_options(visionflow_options INTERFACE -fsanitize=${sanitizer} -fno-omit-frame-pointer)
    target_link_options(visionflow_options INTERFACE -fsanitize=${sanitizer})
  endif()
endforeach()
add_library(visionflow_protocol src/protocol/protocol.cpp)
target_include_directories(visionflow_protocol PUBLIC include)
target_link_libraries(visionflow_protocol PUBLIC visionflow_options Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/unit/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE visionflow_protocol)
  add_test(NAME core COMMAND core_tests)
  set_tests_properties(core PROPERTIES TIMEOUT 20)
endif()
if(VISIONFLOW_CORE_ONLY)
  return()
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(nlohmann_json 3.2 REQUIRED)
add_library(visionflow src/core/config.cpp src/core/logger.cpp src/vision/processor.cpp)
target_include_directories(visionflow PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(visionflow PUBLIC visionflow_protocol nlohmann_json::nlohmann_json ${OpenCV_LIBS})
add_executable(vision_cli apps/vision_cli.cpp)
target_link_libraries(vision_cli PRIVATE visionflow)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(vision_server apps/vision_server.cpp src/net/server.cpp)
  target_link_libraries(vision_server PRIVATE visionflow)
  add_executable(vision_supervisor apps/vision_supervisor.cpp)
  target_link_libraries(vision_supervisor PRIVATE visionflow_protocol)
else()
  message(STATUS "epoll server and POSIX supervisor require Linux/WSL")
endif()
if(BUILD_TESTING)
  add_executable(vision_tests tests/unit/vision_tests.cpp)
  target_link_libraries(vision_tests PRIVATE visionflow)
  add_test(NAME vision COMMAND vision_tests)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_SOURCE_DIR}/tests/integration/test_server.py --bin-dir ${CMAKE_RUNTIME_OUTPUT_DIRECTORY})
    set_tests_properties(integration PROPERTIES TIMEOUT 90)
  endif()
endif()
